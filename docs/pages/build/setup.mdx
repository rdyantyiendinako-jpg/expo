import React, { useState } from 'react';
import { 
  StyleSheet, 
  Text, 
  View, 
  TextInput, 
  Button, 
  FlatList, 
  TouchableOpacity 
} from 'react-native';

// A simple component to render each item in the list
const Item = ({ item, onDelete }) => (
  <View style={styles.itemContainer}>
    <Text style={styles.itemText}>{item.text}</Text>
    {/* The TouchableOpacity makes the 'Delete' text behave like a button */}
    <TouchableOpacity onPress={() => onDelete(item.id)} style={styles.deleteButton}>
      <Text style={styles.deleteButtonText}>Delete</Text>
    </TouchableOpacity>
  </View>
);

// The main application component
export default function App() {
  // 1. State for the list of items. Initial state is an array with one example item.
  const [items, setItems] = useState([
    { id: '1', text: 'Business Card Order' },
    { id: '2', text: 'Social Media Post Draft' },
    { id: '3', text: 'Check Inventory' },
  ]);

  // 2. State for the text currently being typed in the input field.
  const [itemInput, setItemInput] = useState('');

  // 3. Function to handle adding a new item
  const addItemHandler = () => {
    if (itemInput.trim().length === 0) {
      // Don't add if the input is empty
      return; 
    }
    
    // Use the functional form of setItems to ensure we're working with the latest state
    setItems(currentItems => [
      // Add the new item to the beginning of the list
      { id: Math.random().toString(), text: itemInput.trim() }, 
      ...currentItems, 
    ]);
    
    // Clear the input field after adding
    setItemInput(''); 
  };

  // 4. Function to handle deleting an item
  const deleteItemHandler = (itemId) => {
    // Filter out the item that matches the passed ID
    setItems(currentItems => {
      return currentItems.filter(item => item.id !== itemId);
    });
  };

  return (
    <View style={styles.screen}>
      <Text style={styles.title}>Simple Business Task Manager</Text>
      
      {/* Input section to add new items */}
      <View style={styles.inputContainer}>
        <TextInput
          placeholder="Enter new task or item"
          style={styles.input}
          onChangeText={setItemInput} // Update itemInput state as the user types
          value={itemInput} // Control the text displayed in the input
        />
        <Button 
          title="ADD" 
          onPress={addItemHandler} 
          color="#1E90FF" // Dodger Blue
        />
      </View>

      {/* List section */}
      <FlatList 
        data={items} // The data source (the items state array)
        renderItem={({ item }) => (
          // Render the custom Item component for each item in the array
          <Item item={item} onDelete={deleteItemHandler} />
        )}
        keyExtractor={item => item.id} // Tells FlatList how to get a unique key for each item
      />
    </View>
  );
}
